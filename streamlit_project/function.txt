2. text input :  user enter the  string


name =st.text_input("What is your name?")

    st.write("Hello ", name)

3. number  input :  user enter the number
age =st.number_input(
    "Enter the age",
    min_value=1,
    max_value=100

    )

st.write("your age is : ",age)

4.slider : user allow to select a value by moving the slider
salary =st.slider(
    "my expected salary",
    min_value=10000,
    max_value=100000,
    step=1000
)

st.write("your salary is : ",salary)

5.select box /dropdown : dropdown menu
city = st.selectbox(
    "select city",
    ["New York", "Los Angeles", "Chicago", "Houston", "Phoenix", "Philadelphia"]
)
st.write("you selected city is ", city)

6.radio button : user can select one of the options

gender =st.radio("select gender",["male","female"])
st.write("you selected gender is ", gender)
